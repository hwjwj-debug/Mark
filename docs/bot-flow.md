---

## 1) Главное меню (/start)

```mermaid
flowchart TD
  A[/\/start/] --> B{Главное меню}

  B --> M1[MARKET ACC]
  B --> M2[SM MAKRET]
  B --> M3[VPN]
  B --> M4[PROXY]
  B --> P0[Профиль]

flowchart TD
  P0[Профиль] --> P1[Сообщение:\nID + Баланс + Имя]
  P1 --> Pmenu{Кнопки профиля}

  Pmenu --> TopUp[Пополнить баланс]
  Pmenu --> Transfer[Перевести баланс]
  Pmenu --> Ref[Реферальная система]
  Pmenu --> Promo[Активировать промокод]
  Pmenu --> History[История пополнений]

flowchart TD
  TopUp[Пополнить баланс] --> T1[Запрос: введите сумму пополнения]
  T1 --> T2[Пользователь вводит сумму]
  T2 --> T3{Выбор способа пополнения}

  T3 --> Tcb[CryptoBot (API)]
  T3 --> Tcrypto[Криптовалюта (API)]

  Tcb --> Tdone[Создать платёж / инвойс\n→ ожидать подтверждение\n→ начислить баланс]
  Tcrypto --> Tdone

flowchart TD
  Transfer[Перевести баланс] --> TR1[Запрос: введите @username получателя]
  TR1 --> TR2[Пользователь вводит @username]
  TR2 --> TR3{Проверка БД:\nпользователь существует?}

  TR3 -- Нет --> TR404[Сообщение: пользователь не найден]
  TR3 -- Да --> TR4[Запрос: введите сумму перевода]
  TR4 --> TR5[Пользователь вводит сумму]

  TR5 --> TR6[Сообщение: Подтвердите перевод средств]
  TR6 --> TR7{Кнопки}

  TR7 --> TRconfirm[Подтвердить]
  TR7 --> TRcancel[Отменить перевод]

  TRcancel --> TRcmsg[Сообщение: Перевод отменен]

  TRconfirm --> TR8{Проверка баланса в БД:\nхватает средств?}
  TR8 -- Нет --> TRno[Сообщение: недостаточно средств]
  TR8 -- Да --> TR9[Транзакция:\n- списать у отправителя\n- зачислить получателю]
  TR9 --> TRok[Сообщение: Перевод выполнен]

flowchart TD
  M1[MARKET ACC] --> C1[Сообщение: Выбрать категорию товара]
  C1 --> C2{Категории}
  C2 --> CatDRAW[DRAW]
  C2 --> CatFORPOST[FORPOST]

flowchart TD
  CatDRAW[DRAW] --> L1[Сообщение:\nКатегория: DRAW\nОписание категории]
  CatFORPOST[FORPOST] --> L2[Сообщение:\nКатегория: FORPOST\nОписание категории]

  L1 --> Buttons1{Кнопки в категории}
  L2 --> Buttons2{Кнопки в категории}

  Buttons1 --> Item1[Товар 1: DRAW - N$ - Nшт]
  Buttons1 --> Item2[Товар 2: DRAW2 - N$ - Nшт]
  Buttons1 --> Fav1[Добавить в избранное]
  Buttons1 --> BackCats1[Назад ко всем категориям]

  Buttons2 --> Item3[Товар 1: FORPOST - N$ - Nшт]
  Buttons2 --> Item4[Товар 2: FORPOST2 - N$ - Nшт]
  Buttons2 --> Fav2[Добавить в избранное]
  Buttons2 --> BackCats2[Назад ко всем категориям]

  BackCats1 --> C1
  BackCats2 --> C1

flowchart TD
  ItemX[Нажатие на товар] --> Card[Карточка товара:\nТовар + Цена + Описание]

  Card --> A1{Кнопки}
  A1 --> Qty[Выбрать количество]
  A1 --> BackAll[Назад ко всем категориям]

  BackAll --> C1

  Qty --> Q1[Запрос: введите количество]
  Q1 --> Q2[Пользователь вводит количество]

  Q2 --> Q3{Проверка наличия:\nесть нужное кол-во?}
  Q3 -- Нет --> Qno[Сообщение: нет в наличии]
  Q3 -- Да --> Q4{Проверка баланса:\nхватает средств?}

  Q4 -- Нет --> Qnom[Сообщение: недостаточно средств]
  Q4 -- Да --> Q5[Сообщение подтверждения:\nТовар + Кол-во + Итоговая цена]

  Q5 --> Qbtn{Кнопки}
  Qbtn --> BuyOK[Подтвердить]
  Qbtn --> BuyCancel[Отменить покупку]

  BuyCancel --> Bcmsg[Сообщение: покупка отменена]
  BuyOK --> BuyTx[Выдать товар\n+ списать сумму с баланса]
  BuyTx --> Bokmsg[Сообщение: покупка успешна]
